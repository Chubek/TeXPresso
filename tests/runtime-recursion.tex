\def\TP@fn@f#1{%
  \TP@enter
  \TP@aux=#1\relax
  \TP@copycaller{1}{\the\TP@aux}
  \ifnum\TP@load{1}=0\relax
    \global\TP@result=1\relax
  \else
    \global\TP@result=\TP@load{1}\relax
    \global\advance\TP@result by-1\relax
    \global\TP@tag=1\relax
    \TP@store{2}{\TP@result}
    \TP@fn@f{2}
    \TP@store{3}{\TP@result}
    \global\TP@result=\TP@load{1}\relax
    \global\multiply\TP@result by\TP@load{3}\relax
  \fi
  \global\TP@tag=1\relax
  \TP@leave
}
