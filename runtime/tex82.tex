% TeXPresso numeric Scheme runtime, compiled for TeX82/Plain TeX.
% No LaTeX, pages, fonts, boxes, or \shipout.
% This runtime is prepended to compiler output by the file-I/O launcher.
\catcode`\@=11
\escapechar=-1
\newcount\TP@frame
\newcount\TP@result
\newcount\TP@tag
\newcount\TP@aux
\newcount\TP@truthvalue
\TP@frame=0
\TP@result=0
\TP@tag=1
\def\TP@store#1#2{%
  \expandafter\xdef\csname TP@L@\the\TP@frame:#1\endcsname{\number#2}%
  \expandafter\xdef\csname TP@T@\the\TP@frame:#1\endcsname{\the\TP@tag}%
}
\def\TP@load#1{\csname TP@L@\the\TP@frame:#1\endcsname}
\def\TP@loadtag#1{\csname TP@T@\the\TP@frame:#1\endcsname}
\def\TP@caller#1{\csname TP@L@\TP@load{0}:#1\endcsname}
\def\TP@callertag#1{\csname TP@T@\TP@load{0}:#1\endcsname}
\def\TP@copycaller#1#2{%
  \expandafter\xdef\csname TP@L@\the\TP@frame:#1\endcsname{\TP@caller{#2}}%
  \expandafter\xdef\csname TP@T@\the\TP@frame:#1\endcsname{\TP@callertag{#2}}%
}
\def\TP@gstore#1#2{%
  \expandafter\xdef\csname TP@G@#1\endcsname{\number#2}%
  \expandafter\xdef\csname TP@GT@#1\endcsname{\the\TP@tag}%
}
\def\TP@gload#1{\csname TP@G@#1\endcsname}
\def\TP@gtagload#1{\csname TP@GT@#1\endcsname}
\def\TP@enter{%
  \TP@aux=\TP@frame\relax
  \global\advance\TP@frame by1\relax
  \TP@store{0}{\TP@aux}%
}
\def\TP@leave{\global\advance\TP@frame by-1\relax}
\def\TP@settruth{%
  \TP@truthvalue=1\relax
  \ifnum\TP@tag=2\relax
    \ifnum\TP@result=0\relax\TP@truthvalue=0\relax\fi
  \fi
}
\def\TP@requireint#1{%
  \ifnum\TP@loadtag{#1}=1\relax\else
    \errmessage{TeXPresso type error: expected integer}%
  \fi
}
\def\TP@checkdivisor#1{\ifnum\TP@load{#1}=0\relax\errmessage{TeXPresso: division by zero}\fi}
\def\TP@println{%
  \ifnum\TP@tag=2\relax
    \ifnum\TP@result=0\relax
      \immediate\write16{\string\#f}%
    \else
      \immediate\write16{\string\#t}%
    \fi
  \else
    \immediate\write16{\the\TP@result}%
  \fi
}
\def\TP@display{%
  \ifnum\TP@tag=2\relax
    \ifnum\TP@result=0\relax\message{\string\#f}%
    \else\message{\string\#t}%
    \fi
  \else\message{\the\TP@result}%
  \fi
}
